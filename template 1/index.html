# deploy payments-api to three regions
service: payments-api
trigger:
  branch: main
stages:
  - build:  { image: node:22, cache: true }
  - test:   { parallel: 8 }
  - scan:   { fail_on: critical }
  - deploy:
      strategy: canary
      steps: [10, 50, 100]
      regions: [ap-south-1, eu-west-1, us-east-1]
      rollback_if:
        p99_ms: "> 250"
        error_rate: "> 0.1%"
